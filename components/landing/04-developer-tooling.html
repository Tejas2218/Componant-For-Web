deploy.config.ts
import { defineCluster, PostgresDB } from '@clikit/sdk';

// Define global serverless database
export const db = new PostgresDB({
  name: 'hackathon-prod-db',
  region: 'us-east-1',
  autoScale: true,
});

// Deploy cluster automatically
await defineCluster({ db });
